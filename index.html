Needs one secret: ANTHROPIC_API_KEY import { createClient } from 'jsr:@supabase/supabase-js@2'; const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', 'Access-Control-Allow-Methods': 'POST, OPTIONS', }; const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors, 'Content-Type': 'application/json' } }); Deno.serve(async (req) => { if (req.method === 'OPTIONS') return new Response('ok', { headers: cors }); // Only signed-in Orbit users can spend AI credit const sb = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_ANON_KEY')!, { global: { headers: { Authorization: req.headers.get('Authorization') ?? '' } }, }); const { data: { user } } = await sb.auth.getUser(); if (!user) return json({ error: 'Sign in required' }, 401); const { path = [], idea = '', want = 'goal', existing = [], levelsLeft = 3 } = await req.json().catch(() => ({})); const clip = (x: unknown) => String(x).slice(0, 120); let ask: string; if (want === 'plan') { ask = `The person wants to: "${clip(idea)}". Return exactly one item of type "goal" whose title is a clear, motivating version of that goal. Give it 3 to 5 sub-goals (type "goal"), each with 2 to 4 tasks (type "task") as children.`; } else if (want === 'task') { ask = `Goal path: ${path.map(clip).join(' > ')}. Suggest 6 to 8 tasks (type "task") that directly move the last goal forward. No children.`; } else { ask = `Goal path: ${path.map(clip).join(' > ')}. Suggest 5 to 7 sub-goals (type "goal") for the last goal in the path.` + (levelsLeft > 1 ? ' Give each one 2 to 4 tasks (type "task") as children.' : ' No children.'); } const prompt = `You help people break life goals into practical plans. ${ask} Rules: sub-goals are outcomes, specific and measurable where it makes sense. Tasks are single concrete actions someone could finish in a day or less, starting with a verb. Titles under 60 characters, plain language, no numbering, no emoji. Do not repeat these existing items: ${JSON.stringify((existing as unknown[]).slice(0, 40).map(clip))}. Respond with only JSON and nothing else, in this shape: {"items":[{"title":"...","type":"goal","children":[{"title":"...","type":"task"}]}]}`; const r = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'x-api-key': Deno.env.get('ANTHROPIC_API_KEY')!, 'anthropic-version': '2023-06-01', 'content-type': 'application/json', }, body: JSON.stringify({ model: 'claude-haiku-4-5-20251001', max_tokens: 1500, messages: [{ role: 'user', content: prompt }] }), }); if (!r.ok) return json({ error: `AI request failed (${r.status})` }, 502); const data = await r.json(); const text = (data.content ?? []).map((c: { text?: string }) => c.text ?? '').join(''); try { return json(JSON.parse(text.slice(text.indexOf('{'), text.lastIndexOf('}') + 1))); } catch { return json({ error: 'Could not read the AI response' }, 502); } });
